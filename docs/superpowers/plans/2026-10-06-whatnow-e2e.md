# whatnow e2e (5d) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Browser tests that run against the real server, built with a rule-based fake model, so the spec's three flows and the hand-checked UI have regression guards.

**Architecture:** `internal/llm/fake` answers every model call by fixed rules, reading task numbers from the real prompts. A file with the `e2e` build tag swaps it in for the real client and mocks the keychain. Playwright in `web/e2e/` drives the built binary, wiping the database before each test.

**Tech Stack:** Go 1.27, go-keyring `MockInit`, `@playwright/test` (Chromium), SvelteKit static build, `just`.

**Spec:** `docs/superpowers/specs/2026-10-06-whatnow-e2e-design.md`

All code below was compiled and run in a scratch worktree: Go tests green, and the Playwright suite passed 9/9 on five runs in a row (about 6s each).

## Global Constraints

- Tests never call a real model or the network, and never touch the real keychain.
- The normal binary (no tag) behaves as before.
- `just test` stays browser-free. Playwright runs only from `just e2e`.
- One Playwright worker, Chromium only.
- npm installs go through `web/.npmrc` (registry.npmjs.org). Run npm from `web/`.

## Review Focus

- **The shell has `OPENAI_API_KEY` or `ANTHROPIC_API_KEY` set.** The e2e binary unsets both in `init`, so onboarding still asks for a key and no real key ever reaches a client. The onboarding test fails if this regresses (it would show "Found your key in…").
- **A note that's only separators** ("and", ",,"). The fake keeps the raw note as the title rather than sending an empty one, which would fail the cleanup. This is pinned in Task 1's table test.
- **A prompt format change** (the eligible-task heading, or the "  - blocked by #" annotation). The drift guards in Task 1 fail with a fallback plan or a fallback pick before Playwright ever runs.
- **Port 7788 already in use.** Playwright's `webServer` refuses to start against a port that answers. Expected: a clear startup error, not tests running against a stale server.
- **Toasts expiring mid-test.** Tests click Undo and read toasts right after they appear. If the suite flakes here, raise the toast assertion timeout instead of adding sleeps.

func TestCleanupSplitsANoteIntoTitles(t *testing.T) {
	tests := []struct {
		raw  string
		want []string
	}{
		{"buy milk", []string{"buy milk"}},
		{"buy milk and call mom", []string{"buy milk", "call mom"}},
		{"buy milk, call mom\nwater the plants", []string{"buy milk", "call mom", "water the plants"}},
		{"sand the candy dish", []string{"sand the candy dish"}},
		{" buy milk ,, and ", []string{"buy milk"}},
		{"and", []string{"and"}},
	}
	for _, tt := range tests {
		t.Run(tt.raw, func(t *testing.T) {
			reply, err := fake.Client{}.Complete(context.Background(), llm.Request{SchemaName: "cleanup", User: tt.raw})
			if err != nil {
				t.Fatal(err)
			}
			var got struct {
				Original struct {
					Keep  bool   `json:"keep"`
					Title string `json:"title"`
				} `json:"original"`
				NewTasks []struct {
					Ref   string `json:"ref"`
					Title string `json:"title"`
				} `json:"new_tasks"`
			}
			if err := json.Unmarshal([]byte(reply), &got); err != nil {
				t.Fatal(err)
			}
			if !got.Original.Keep {
				t.Error("original not kept")
			}
			titles := []string{got.Original.Title}
			for i, n := range got.NewTasks {
				if want := "n" + string(rune('1'+i)); n.Ref != want {
					t.Errorf("ref %d = %q, want %q", i, n.Ref, want)
				}
				titles = append(titles, n.Title)
			}
			if diff := cmp.Diff(tt.want, titles); diff != "" {
				t.Errorf("titles (-want +got):\n%s", diff)
			}
		})
	}
}

// Package fake is a model that answers by fixed rules, for the end-to-end tests. It reads task
// numbers out of the prompt, so its replies pass the same checks a real model's must.
package fake

import (
	"context"
	"encoding/json"
	"regexp"
	"strconv"
	"strings"

	"github.com/jameynakama/whatnow/internal/llm"
)

// The fixed text the fake answers with. The Playwright tests look for these.
const (
	Opening   = "A fake plan, in order."
	Rationale = "Next in line."
	Reason    = "First free task."
	Suggest   = "Keep it going."
	Sentence  = "A fake summary."
	ChatReply = "A fake reply. Nothing changed."
)

// Client answers Complete by the request's schema name, and Chat with ChatReply.
type Client struct{}

func (Client) Complete(_ context.Context, req llm.Request) (string, error) {
	var reply any
	switch req.SchemaName {
	case "cleanup":
		reply = cleanup(req.User)
	case "plan":
		reply = plan(eligible(system(req.System)))
	case "pick":
		reply = pick(eligible(system(req.System)), req.User)
	case "suggestions":
		reply = suggestions(req.User)
	default:
		return Sentence, nil
	}
	b, err := json.Marshal(reply)
	return string(b), err
}

// Chat streams ChatReply a word at a time and never calls a tool.
func (Client) Chat(_ context.Context, _ llm.ChatRequest, onText func(string)) (llm.Reply, error) {
	for _, word := range strings.SplitAfter(ChatReply, " ") {
		onText(word)
	}
	return llm.Reply{Text: ChatReply}, nil
}

func system(blocks []llm.Block) string {
	var b strings.Builder
	for _, blk := range blocks {
		b.WriteString(blk.Text)
	}
	return b.String()
}

var splitter = regexp.MustCompile(`,|\band\b|\n`)

// cleanup keeps the note's first piece as the original and makes each other piece a new task.
func cleanup(raw string) map[string]any {
	var pieces []string
	for _, p := range splitter.Split(raw, -1) {
		if p = strings.TrimSpace(p); p != "" {
			pieces = append(pieces, p)
		}
	}
	if len(pieces) == 0 {
		pieces = []string{strings.TrimSpace(raw)}
	}
	original := task(pieces[0])
	original["keep"] = true
	newTasks := []map[string]any{}
	for i, p := range pieces[1:] {
		t := task(p)
		t["ref"] = "n" + strconv.Itoa(i+1)
		newTasks = append(newTasks, t)
	}
	return map[string]any{
		"original":  original,
		"new_tasks": newTasks,
		"events":    []any{},
		"blockers":  []any{},
	}
}

func task(title string) map[string]any {
	return map[string]any{
		"title": title, "notes": "", "size": "", "energy": "", "feeling": "",
		"due_on": "", "not_before": "", "refs": []any{},
	}
}

type fact struct {
	id      int64
	blocked bool
}

const eligibleHeading = "Eligible tasks, as #<id> title:\n"

var taskLine = regexp.MustCompile(`^#(\d+) `)

// eligible reads the facts' task list: "#<id> title" lines, each followed by indented notes such
// as "  - blocked by #3". The list ends at the first line that is neither.
func eligible(prompt string) []fact {
	_, list, _ := strings.Cut(prompt, eligibleHeading)
	var facts []fact
	for _, line := range strings.Split(list, "\n") {
		if m := taskLine.FindStringSubmatch(line); m != nil {
			id, _ := strconv.ParseInt(m[1], 10, 64)
			facts = append(facts, fact{id: id})
			continue
		}
		if !strings.HasPrefix(line, "  ") || len(facts) == 0 {
			break
		}
		if strings.HasPrefix(line, "  - blocked by #") {
			facts[len(facts)-1].blocked = true
		}
	}
	return facts
}

// plan tiers tasks in the order the facts list them: a must, a should, then coulds.
func plan(facts []fact) map[string]any {
	tiers := []string{"must", "should"}
	entries := []map[string]any{}
	for i, f := range facts {
		tier := "could"
		if i < len(tiers) {
			tier = tiers[i]
		}
		entries = append(entries, map[string]any{"task_id": f.id, "rationale": Rationale, "tier": tier})
	}
	return map[string]any{"entries": entries, "opening": Opening}
}

var excluded = regexp.MustCompile(`Not #(\d+),`)

// pick is the first task that isn't blocked and isn't the one the user just passed on.
func pick(facts []fact, user string) map[string]any {
	var skip int64
	if m := excluded.FindStringSubmatch(user); m != nil {
		skip, _ = strconv.ParseInt(m[1], 10, 64)
	}
	var id int64
	for _, f := range facts {
		if !f.blocked && f.id != skip {
			id = f.id
			break
		}
	}
	return map[string]any{"reason": Reason, "task_id": id}
}

var taskRef = regexp.MustCompile(`#(\d+)`)

// suggestions rolls over every task the review asks about.
func suggestions(user string) map[string]any {
	_, asked, _ := strings.Cut(user, "unfinished task:")
	out := []map[string]any{}
	for _, m := range taskRef.FindAllStringSubmatch(asked, -1) {
		id, _ := strconv.ParseInt(m[1], 10, 64)
		out = append(out, map[string]any{"task_id": id, "reason": Suggest, "action": "roll_over", "until": ""})
	}
	return map[string]any{"suggestions": out}
}

package main

import (
